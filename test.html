// ========================================== // MCQ QUIZ SYSTEM // 15 RANDOM QUESTIONS // 1 MINUTE PER QUESTION // ========================================== const QUESTIONS_PER_QUIZ = 15; const TIME_PER_QUESTION = 60; let selectedSubject = ""; let selectedClass = ""; let quizQuestions = []; let currentQuestion = 0; let answers = []; let skipped = []; let timerInterval = null; let timeLeft = TIME_PER_QUESTION; let quizStartTime = 0; let totalTimeUsed = 0; // ========================================== // SCREEN CONTROL // ========================================== function showScreen(id) { document.querySelectorAll(".screen").forEach(screen => { screen.classList.remove("active"); }); const screen = document.getElementById(id); if (screen) { screen.classList.add("active"); } } // ========================================== // SUBJECT SELECTION // ========================================== function selectSubject(subject) { selectedSubject = subject; let name = subject; if (subject === "general") { name = "GENERAL SCIENCE"; } if (subject === "math") { name = "MATHEMATICS"; } if (subject === "social") { name = "SOCIAL SCIENCE"; } if (subject === "english") { name = "ENGLISH"; } if (subject === "all") { name = "ALL SUBJECTS"; } const title = document.getElementById("classTitle"); if (title) { title.textContent = "Choose Class - " + name; } showScreen("classScreen"); } // ========================================== // CLASS SELECTION // ========================================== function selectClass(classNumber) { selectedClass = String(classNumber); startQuiz(); } // ========================================== // BACK TO SUBJECTS // ========================================== function goToSubjects() { stopTimer(); showScreen("subjectScreen"); } // ========================================== // GET QUESTION ARRAY // ========================================== function getArray(variableName) { if (Array.isArray(variableName)) { return variableName; } return []; } // ========================================== // GET QUESTION POOL // ========================================== function getQuestionPool() { let pool = []; // GENERAL SCIENCE if (selectedSubject === "general") { pool = getArray( typeof scienceQuestions !== "undefined" ? scienceQuestions : [] ); } // MATHEMATICS else if (selectedSubject === "math") { pool = getArray( typeof mathQuestions !== "undefined" ? mathQuestions : [] ); } // SOCIAL SCIENCE else if (selectedSubject === "social") { pool = getArray( typeof socialQuestions !== "undefined" ? socialQuestions : [] ); } // ENGLISH else if (selectedSubject === "english") { pool = getArray( typeof englishQuestions !== "undefined" ? englishQuestions : [] ); } // ALL SUBJECTS else if (selectedSubject === "all") { if (typeof mathQuestions !== "undefined") { pool = pool.concat(getArray(mathQuestions)); } if (typeof scienceQuestions !== "undefined") { pool = pool.concat(getArray(scienceQuestions)); } if (typeof socialQuestions !== "undefined") { pool = pool.concat(getArray(socialQuestions)); } if (typeof englishQuestions !== "undefined") { pool = pool.concat(getArray(englishQuestions)); } } // ======================================== // CLASS FILTER // ======================================== pool = pool.filter(question => { if (!question) { return false; } const qClass = question.class ?? question.className ?? question.grade; return String(qClass) === String(selectedClass); }); return pool; } // ========================================== // SHUFFLE // ========================================== function shuffle(array) { const arr = [...array]; for (let i = arr.length - 1; i > 0; i--) { const j = Math.floor( Math.random() * (i + 1) ); [arr[i], arr[j]] = [arr[j], arr[i]]; } return arr; } // ========================================== // START QUIZ // ========================================== function startQuiz() { const pool = getQuestionPool(); if (!pool.length) { alert( "No questions found for " + selectedSubject + " - Class " + selectedClass ); return; } quizQuestions = shuffle(pool).slice( 0, Math.min( QUESTIONS_PER_QUIZ, pool.length ) ); currentQuestion = 0; answers = new Array(quizQuestions.length) .fill(null); skipped = new Array(quizQuestions.length) .fill(false); quizStartTime = Date.now(); totalTimeUsed = 0; showScreen("quizScreen"); updateQuizHeader(); updateAnalytics(); loadQuestion(); startTimer(); } // ========================================== // QUESTION TEXT // ========================================== function getQuestionText(question) { return ( question.question ?? question.text ?? question.q ?? "Question not found" ); } // ========================================== // OPTIONS // ========================================== function getOptions(question) { if (Array.isArray(question.options)) { return question.options; } if (Array.isArray(question.answers)) { return question.answers; } return [ question.optionA, question.optionB, question.optionC, question.optionD ].filter(x => x !== undefined); } // ========================================== // CORRECT ANSWER // ========================================== function getCorrectAnswer(question) { return ( question.answer ?? question.correctAnswer ?? question.correct ?? question.correctOption ); } // ========================================== // LOAD QUESTION // ========================================== function loadQuestion() { const question = quizQuestions[currentQuestion]; if (!question) { finishQuiz(); return; } const questionText = document.getElementById("questionText"); const optionsContainer = document.getElementById("optionsContainer"); questionText.textContent = getQuestionText(question); optionsContainer.innerHTML = ""; const options = getOptions(question); const savedAnswer = answers[currentQuestion]; options.forEach((option, index) => { const button = document.createElement("button"); button.className = "option-btn"; button.textContent = option; if (savedAnswer === index) { button.classList.add("selected"); } button.onclick = function () { selectAnswer(index); }; optionsContainer.appendChild(button); }); updateQuizHeader(); updateAnalytics(); resetQuestionTimer(); } // ========================================== // SELECT ANSWER // ========================================== function selectAnswer(index) { answers[currentQuestion] = index; skipped[currentQuestion] = false; const buttons = document.querySelectorAll( "#optionsContainer .option-btn" ); buttons.forEach((button, i) => { button.classList.toggle( "selected", i === index ); }); updateAnalytics(); } // ========================================== // SKIP QUESTION // ========================================== function skipQuestion() { answers[currentQuestion] = null; skipped[currentQuestion] = true; moveNext(); } // ========================================== // SAVE AND NEXT // ========================================== function saveAndNext() { if (answers[currentQuestion] === null) { skipped[currentQuestion] = true; } else { skipped[currentQuestion] = false; } moveNext(); } // ========================================== // NEXT // ========================================== function nextQuestion() { moveNext(); } // ========================================== // MOVE NEXT // ========================================== function moveNext() { if ( currentQuestion < quizQuestions.length - 1 ) { currentQuestion++; loadQuestion(); } else { finishQuiz(); } } // ========================================== // PREVIOUS QUESTION // ========================================== function previousQuestion() { if (currentQuestion > 0) { currentQuestion--; loadQuestion(); } } // ========================================== // TIMER // ========================================== function startTimer() { stopTimer(); timerInterval = setInterval(() => { timeLeft--; updateTimerDisplay(); if (timeLeft <= 0) { timeLeft = 0; updateTimerDisplay(); autoNextQuestion(); } }, 1000); } // ========================================== // RESET QUESTION TIMER // ========================================== function resetQuestionTimer() { timeLeft = TIME_PER_QUESTION; updateTimerDisplay(); } // ========================================== // AUTO NEXT AFTER 1 MINUTE // ========================================== function autoNextQuestion() { if (answers[currentQuestion] === null) { skipped[currentQuestion] = true; } if ( currentQuestion < quizQuestions.length - 1 ) { currentQuestion++; loadQuestion(); } else { finishQuiz(); } } // ========================================== // STOP TIMER // ========================================== function stopTimer() { if (timerInterval !== null) { clearInterval(timerInterval); timerInterval = null; } } // ========================================== // TIMER DISPLAY // ========================================== function updateTimerDisplay() { const timer = document.getElementById("timer"); if (!timer) { return; } const minutes = Math.floor(timeLeft / 60); const seconds = timeLeft % 60; timer.textContent = String(minutes).padStart(2, "0") + ":" + String(seconds).padStart(2, "0"); } // ========================================== // HEADER // ========================================== function updateQuizHeader() { const subjectName = document.getElementById("subjectName"); const className = document.getElementById("className"); const questionNumber = document.getElementById("questionNumber"); const totalQuestions = document.getElementById("totalQuestions"); let subjectText = selectedSubject; if (selectedSubject === "general") { subjectText = "GENERAL SCIENCE"; } else if (selectedSubject === "math") { subjectText = "MATHEMATICS"; } else if (selectedSubject === "social") { subjectText = "SOCIAL SCIENCE"; } else if (selectedSubject === "english") { subjectText = "ENGLISH"; } else if (selectedSubject === "all") { subjectText = "ALL SUBJECTS"; } subjectName.textContent = subjectText; className.textContent = "Class " + selectedClass; questionNumber.textContent = currentQuestion + 1; totalQuestions.textContent = quizQuestions.length; } // ========================================== // ANALYTICS // ========================================== function updateAnalytics() { let answered = 0; let correct = 0; let wrong = 0; let skippedCount = 0; quizQuestions.forEach( (question, index) => { const answer = answers[index]; if (answer !== null) { answered++; const options = getOptions(question); const correctAnswer = getCorrectAnswer(question); let correctIndex = getCorrectIndex( question, options, correctAnswer ); if (answer === correctIndex) { correct++; } else { wrong++; } } else if (skipped[index]) { skippedCount++; } } ); document.getElementById( "sideQuestion" ).textContent = (currentQuestion + 1) + " / " + quizQuestions.length; document.getElementById( "answeredCount" ).textContent = answered; document.getElementById( "correctCount" ).textContent = correct; document.getElementById( "wrongCount" ).textContent = wrong; document.getElementById( "skippedCount" ).textContent = skippedCount; const elapsed = Math.floor( (Date.now() - quizStartTime) / 1000 ); document.getElementById( "timeUsed" ).textContent = formatTime(elapsed); } // ========================================== // FIND CORRECT ANSWER INDEX // ========================================== function getCorrectIndex( question, options, correctAnswer ) { if ( typeof correctAnswer === "number" ) { if ( correctAnswer >= 0 && correctAnswer < options.length ) { return correctAnswer; } if ( correctAnswer >= 1 && correctAnswer <= options.length ) { return correctAnswer - 1; } } if ( typeof correctAnswer === "string" ) { const answerText = correctAnswer.trim(); // A / B / C / D const letter = answerText.toUpperCase(); if ( ["A", "B", "C", "D"].includes(letter) ) { return ( letter.charCodeAt(0) - "A".charCodeAt(0) ); } // Exact option text const index = options.findIndex( option => String(option).trim() === answerText ); if (index !== -1) { return index; } } return -1; } // ========================================== // FORMAT TIME // ========================================== function formatTime(seconds) { const minutes = Math.floor(seconds / 60); const secs = seconds % 60; return ( minutes + ":" + String(secs).padStart(2, "0") ); } // ========================================== // FINISH QUIZ // ========================================== function finishQuiz() { stopTimer(); totalTimeUsed = Math.floor( (Date.now() - quizStartTime) / 1000 ); let correct = 0; let wrong = 0; let skippedCount = 0; quizQuestions.forEach( (question, index) => { if (answers[index] === null) { skippedCount++; return; } const options = getOptions(question); const correctAnswer = getCorrectAnswer(question); const correctIndex = getCorrectIndex( question, options, correctAnswer ); if ( answers[index] === correctIndex ) { correct++; } else { wrong++; } } ); const total = quizQuestions.length; const percentage = total > 0 ? Math.round( (correct / total) * 100 ) : 0; const qpm = totalTimeUsed > 0 ? ( total / (totalTimeUsed / 60) ).toFixed(2) : total; document.getElementById( "finalMarks" ).textContent = correct + " / " + total; document.getElementById( "finalCorrect" ).textContent = correct; document.getElementById( "finalWrong" ).textContent = wrong; document.getElementById( "finalSkipped" ).textContent = skippedCount; document.getElementById( "finalTime" ).textContent = formatTime(totalTimeUsed); document.getElementById( "questionsPerMinute" ).textContent = qpm; document.getElementById( "resultPercent" ).textContent = percentage + "%"; setResultLevel(correct); let message = "Keep practicing!"; if (correct >= 11) { message = "Excellent! 🔥"; } else if (correct >= 7) { message = "Very Good! 👍"; } else if (correct >= 4) { message = "Good Try! 💪"; } document.getElementById( "resultMessage" ).textContent = message; showScreen("resultScreen"); } // ========================================== // RESULT COLOR LEVEL // ========================================== function setResultLevel(correct) { const circle = document.getElementById( "resultCircle" ); const level = document.getElementById( "resultLevel" ); // Remove old classes circle.classList.remove( "red", "orange", "yellow", "green" ); if (correct >= 11) { circle.classList.add("green"); level.textContent = "🟢 11–15 Correct"; } else if (correct >= 7) { circle.classList.add("yellow"); level.textContent = "🟡 7–10 Correct"; } else if (correct >= 4) { circle.classList.add("orange"); level.textContent = "🟠 4–6 Correct"; } else { circle.classList.add("red"); level.textContent = "🔴 0–3 Correct"; } } // ========================================== // RESTART QUIZ // ========================================== function restartQuiz() { startQuiz(); } // ========================================== // PAGE START // ========================================== document.addEventListener( "DOMContentLoaded", () => { showScreen("subjectScreen"); } );
